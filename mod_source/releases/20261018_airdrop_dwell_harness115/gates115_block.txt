    assert script_txt.count(_line) == 1, "the round-115 constant is not defined exactly once: " + _k

        "the round-115 helper is missing: " + _fn

    "the first evaluation of a contact can fire - the dwell is not enforced"

    "the separation branch lost the edge comparison"

    "the short-crossing fallback is missing"
